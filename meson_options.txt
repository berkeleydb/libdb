# Build options for the Meson build of libdb.
#
# PBT (property-based testing) is opt-in and OFF by default so the core
# library build never depends on hegel-c or the `hegel` server binary.
option('hegel', type: 'feature', value: 'disabled',
  description: 'Build the Hegel property-based tests under test/pbt (needs hegel-c + the `hegel` server binary; pip install hegel-core).')

# Mirrors autoconf's --without-uring. The io_uring backend adds a hard
# DT_NEEDED liburing.so.2 to libdb, so declining it has to be possible for
# packagers targeting runtimes whose seccomp profile blocks io_uring syscalls.
# 'enabled' is a hard requirement rather than a preference: it fails the
# configure instead of silently falling back to synchronous I/O.
option('uring', type: 'feature', value: 'auto',
  description: 'Linux io_uring AIO backend for the buffer pool')

# Mirrors autoconf's --enable-typed.  OFF by default; 'disabled' must stay the
# default or run_meson_autoconf_parity.sh's HAVE_* comparison fails.
option('typed', type: 'feature', value: 'disabled',
  description: 'Typed-data and constraint layer (RFC 0014)')

# Mirrors autoconf's --enable-dtrace in NAME ONLY for now: the DTrace/USDT
# build (dtrace -h header, dtrace -G post-link over every object) is not yet
# wired into meson.  'enabled' fails the configure rather than building a
# library without the probes; use dist/configure --enable-dtrace.
option('dtrace', type: 'feature', value: 'disabled',
  description: 'bdb DTrace/USDT probes (not yet implemented in meson; use autoconf --enable-dtrace)')

# RFC 0019 TPM 2.0 key provider.  Off by default so the default build gains no
# dependency.  'enabled' requires libtss2 ESAPI (dependency('tss2-esys')).
option('tpm', type: 'feature', value: 'disabled',
  description: 'TPM 2.0 encryption key provider (needs libtss2 ESAPI)')

# Mirrors autoconf's --enable-tls.  OFF by default so the core library build
# never depends on OpenSSL.  When 'enabled', TLS for the Replication Manager is
# compiled in and libssl/libcrypto become a hard requirement.
option('tls', type: 'feature', value: 'disabled',
  description: 'TLS (OpenSSL) support for the Replication Manager (RFC 0018)')
